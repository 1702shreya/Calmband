# Convert EDA to GSR if present
if 'EDA' in df.columns:
    df['GSR'] = (df['EDA'] - df['EDA'].min()) / (df['EDA'].max() - df['EDA'].min()) * 1023
    df['GSR'] = df['GSR'].round().astype(int)

features = ['GSR', 'HR', 'TEMP']

# Class distribution

# Scale features

# Train-test split

    X_scaled, y, test_size=0.2, random_state=42, stratify=y
)

# Balance classes using SMOTE
smote = SMOTE(random_state=42)
X_train_res, y_train_res = smote.fit_resample(X_train, y_train)
print("\nClass distribution after SMOTE:")
print(Counter(y_train_res))

# LightGBM Model
model = LGBMClassifier(
    objective='multiclass',
    num_class=3,
    boosting_type='gbdt',
    learning_rate=0.05,
    n_estimators=200,
    random_state=42,
    class_weight='balanced'
)

# Train
model.fit(X_train_res, y_train_res)

# Evaluate
y_test_pred = model.predict(X_test)
print("\nTest Accuracy:", accuracy_score(y_test, y_test_pred))
print("\nClassification Report:\n", classification_report(y_test, y_test_pred))

<script>
// needle element
const needle = document.getElementById("needle");

// Map stress (0/1/2 → percent)
function stressToPercent(stress) {
  if (typeof stress === "number" && stress >= 0 && stress <= 100) return stress;
  if (stress === 0) return 0;
  if (stress === 1) return 50;
  if (stress === 2) return 100;
  return 0;
}

// Update needle
function updateNeedleFromPercent(percent) {
  percent = Math.min(100, Math.max(0, percent));
  const angle = 90 - (percent * 1.8);
  needle.setAttribute("transform", `rotate(${angle} 150 140)`);
}

// Update status text
function updateStressStatus(stress) {
  const status = document.getElementById("stressStatus");

  if (stress === 0) {
    status.textContent = "✅ Your Stress is LOW";
    status.style.color = "#00ff99";
    status.style.textShadow = "0 0 18px #00ff99";
  } 
  else if (stress === 1) {
    status.textContent = "⚠️ Your Stress is MID";
    status.style.color = "#ffff55";
    status.style.textShadow = "0 0 18px #ffff55";
  } 
  else if (stress === 2) {
    status.textContent = "🔥 Your Stress is HIGH";
    status.style.color = "#ff4444";
    status.style.textShadow = "0 0 18px #ff4444";
  }
}

// ✅ FETCH DATA DIRECTLY FROM ARDUINO EVERY 300ms
async function fetchArduinoData() {
  try {
    const res = await fetch("http://10.139.50.176/data");
    const data = await res.json();

    console.log("ESP DATA:", data);

    // Update gauge
    const percent = stressToPercent(data.STRESS);
    updateNeedleFromPercent(percent);
    updateStressStatus(data.STRESS);

    // Update cards
    document.querySelector(".cards .card:nth-child(1)").textContent =
      `Heart Rate: ${data.HR} bpm`;

    document.querySelector(".cards .card:nth-child(2)").textContent =
      `Skin Temperature: ${data.TEMP} °C`;

    document.querySelector(".cards .card:nth-child(3)").textContent =
      `GSR Level: ${data.GSR}`;

  } catch (err) {
    console.log("Error fetching from ESP:", err);
  }
}

setInterval(fetchArduinoData, 300);

// Music control (unchanged)
const musicBtn = document.getElementById('musicBtn');
const relaxMusic = document.getElementById('relaxMusic');
const musicControl = document.getElementById('musicControl');
let isPlaying = false;

relaxMusic.src = "music/peaceful-piano-background-music-423216.mp3";

musicControl.addEventListener('click', (e) => {
  if (e.target.classList.contains('explore-arrow')) return;
  if (isPlaying) {
    relaxMusic.pause();
    musicBtn.textContent = "🎵 Music Off";
    isPlaying = false;
  } else {
    relaxMusic.play();
    musicBtn.textContent = "🎵 Music On";
    isPlaying = true;
  }
});
